cmake_minimum_required(VERSION 3.16)
project(stage1_comm_test)
# Disable project test programs for competition deployment.
set(BUILD_TESTING OFF CACHE BOOL "Competition-only build" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tdt_interface REQUIRED)
find_package(campus_aim_interfaces REQUIRED)
find_package(Eigen3 REQUIRED)

# 按已安装的校园赛通信接口选择会话字段。
file(STRINGS "${tdt_interface_DIR}/../msg/AutoAimFrame.msg" FRAME_SESSION_FIELD REGEX "^[^#]*[ \t]session_id([ \t]|$)")
file(STRINGS "${tdt_interface_DIR}/../msg/AutoAimState.msg" STATE_SESSION_FIELD REGEX "^[^#]*[ \t]session_id([ \t]|$)")
if(FRAME_SESSION_FIELD)
  add_compile_definitions(CAMPUS_FRAME_HAS_SESSION_ID)
endif()
if(STATE_SESSION_FIELD)
  add_compile_definitions(CAMPUS_STATE_HAS_SESSION_ID)
endif()

# 与发行版 cv_bridge 使用同一套 OpenCV，避免不同版本的 ABI 混用。
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPENCV4 REQUIRED opencv4)
pkg_get_variable(OPENCV_LIBRARY_DIR opencv4 libdir)
if(EXISTS "${OPENCV_LIBRARY_DIR}/cmake/opencv4/OpenCVConfig.cmake")
  set(OpenCV_DIR "${OPENCV_LIBRARY_DIR}/cmake/opencv4" CACHE PATH "OpenCV matching distro cv_bridge" FORCE)
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui calib3d dnn videoio)
find_package(cv_bridge REQUIRED)
message(STATUS "Competition OpenCV ${OpenCV_VERSION}: ${OpenCV_DIR}")

# competition_auto.launch.py 实际使用的八个节点。
add_executable(armor_detector src/detection/armor_detector.cpp)
add_executable(tracker src/tracking/tracker.cpp)
add_executable(aim_solver src/aiming/aim_solver.cpp)
add_executable(combat_gimbal src/aiming/combat_gimbal.cpp)
add_executable(buff_controller src/buff_detection/buff_node.cpp)
add_executable(static_grid_publisher src/mapping/static_grid_publisher.cpp)
add_executable(dijkstra_planner src/navigation/dijkstra_planner.cpp)
add_executable(path_follower src/navigation/path_follower.cpp)
set(COMPETITION_NODES armor_detector tracker aim_solver combat_gimbal buff_controller
  static_grid_publisher dijkstra_planner path_follower)
foreach(node IN LISTS COMPETITION_NODES)
  ament_target_dependencies(${node} rclcpp ament_index_cpp geometry_msgs nav_msgs
    visualization_msgs sensor_msgs std_msgs tdt_interface campus_aim_interfaces cv_bridge)
  target_include_directories(${node} PRIVATE src)
  target_include_directories(${node} SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(${node} ${OpenCV_LIBS} Eigen3::Eigen nlohmann_json::nlohmann_json)
endforeach()
install(TARGETS ${COMPETITION_NODES} DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS tools/competition_preflight.py tools/competition_training_profile.py
  tools/autoaim_profile.py tools/capture_competition_status.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config models maps DESTINATION share/${PROJECT_NAME})
ament_package()
